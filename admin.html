05 / Album ảnh tư liệu sự kiện

Có thể chọn nhiều ảnh cùng lúc từ máy tính hoặc gắn đường dẫn URL kèm chú thích.

Hướng dẫn kết nối Cơ sở dữ liệu Supabase (Thành viên & Sự kiện)

Để lưu trữ ảnh và dữ liệu trực tuyến cho mọi người truy cập khi xuất bản website, thực hiện 3 bước sau trên supabase.com (gói miễn phí):

  1. Tạo Project mới, vào mục Authentication → Users → Add User để tạo tài khoản email và mật khẩu quản trị viên.
  2. Mở mục SQL Editor → New Query, dán đoạn mã dưới đây và nhấn Run (tự động tạo bảng members, bảng events và 2 kho ảnh):
-- 1. Bảng thành viên Phòng truyền thống
create table if not exists public.members (
    id text primary key,
    name text not null,
    gen integer not null default 1,
    term text default '',
    tier integer not null default 4,
    sort_order integer not null default 10,
    role text default '',
    department text default '',
    cohort text default '',
    avatar_url text default '',
    quote text default '',
    confession_title text default '',
    confession text default '',
    created_at timestamptz default now()
);

-- 2. Bảng Hoạt động & Sự kiện (Core hằng năm & Ngoài lề)
create table if not exists public.events (
    id text primary key,
    title text not null,
    category text not null default 'core',
    sub_type text default '',
    annual_cycle text default '',
    date_label text default '',
    year integer default 2026,
    location text default '',
    scale text default '',
    is_featured boolean default true,
    sort_order integer default 10,
    cover_url text default '',
    summary text default '',
    highlight_quote text default '',
    content text default '',
    activities jsonb default '[]'::jsonb,
    editions jsonb default '[]'::jsonb,
    gallery jsonb default '[]'::jsonb,
    created_at timestamptz default now()
);

-- 3. Phân quyền bảo mật RLS
alter table public.members enable row level security;
alter table public.events enable row level security;

create policy "Cong khai xem thanh vien" on public.members for select using (true);
create policy "Admin quan ly thanh vien" on public.members for all to authenticated using (true) with check (true);

create policy "Cong khai xem su kien" on public.events for select using (true);
create policy "Admin quan ly su kien" on public.events for all to authenticated using (true) with check (true);

-- 4. Kho lưu trữ ảnh chân dung & ảnh sự kiện
insert into storage.buckets (id, name, public) values ('member-photos', 'member-photos', true) on conflict (id) do nothing;
insert into storage.buckets (id, name, public) values ('event-photos', 'event-photos', true) on conflict (id) do nothing;

create policy "Xem anh cong khai" on storage.objects for select using (bucket_id in ('member-photos', 'event-photos'));
create policy "Admin tai anh len" on storage.objects for insert to authenticated with check (bucket_id in ('member-photos', 'event-photos'));

3. Vào Project Settings → Data API, sao chép Project URL và anon public key điền vào tệp js/supabase-config.js.
